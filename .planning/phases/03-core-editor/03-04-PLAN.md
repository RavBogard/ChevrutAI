         }
       }, [sheetId, currentUser?.uid]);

       const handleSendMessage = (text) => useSheetStore.getState().sendMessage(text);

       const handleNewSheet = () => {
         const newId = Date.now().toString();

         navigate(`/sheet/${newId}`, { state: { isNew: true } });

       const handleDeleteSheet = (id) => useSheetStore.getState().deleteSheet(id, currentUser?.uid);

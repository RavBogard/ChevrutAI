  <action>
    Create src/hooks/useAutosave.js. This hook subscribes to the store and debounces Firestore writes.

    Implementation rules:
    - NO lodash — use hand-rolled debounce with setTimeout/clearTimeout refs (consistent with existing codebase pattern)
    - Use useSheetStore.subscribe() (subscribeWithSelector) to watch title and sources
    - When title or sources change: set isDirty=true, schedule a debounced save
    - Debounce delay: 1000ms (matches existing useSheetPersistence behavior)
    - The debounce must clear on unmount (return cleanup from useEffect)
    - Save calls setIsSaving(true), then saveSheetToFirestore, then setIsSaving(false) + setIsDirty(false) in finally
    - Skip save if no userId or no currentSheetId (guest user — no Firestore write)
    - On save error: log to console, do NOT crash the app (match existing CONCERNS behavior)

    ```javascript
    import { useEffect, useRef } from 'react';
    import useSheetStore from '../stores/useSheetStore';
    import { saveSheetToFirestore } from '../services/firebase';

    export const useAutosave = (userId) => {
      const timerRef = useRef(null);

      useEffect(() => {
        const unsubscribe = useSheetStore.subscribe(
          (state) => ({ title: state.title, sources: state.sources }),
          () => {
            useSheetStore.getState().setIsDirty(true);

            if (timerRef.current) clearTimeout(timerRef.current);

            timerRef.current = setTimeout(async () => {
              const state = useSheetStore.getState();
              if (!userId || !state.currentSheetId) return;

              state.setIsSaving(true);
              try {
                await saveSheetToFirestore(userId, {
                  id: state.currentSheetId,
                  title: state.title,
                  sources: state.sources,
                  schemaVersion: 1,
                });
                useSheetStore.getState().setIsDirty(false);
              } catch (err) {
                console.error('[useAutosave] Firestore save failed:', err);
              } finally {
                useSheetStore.getState().setIsSaving(false);
              }
            }, 1000);
          }
        );

        return () => {
          unsubscribe();
          if (timerRef.current) clearTimeout(timerRef.current);
        };
      }, [userId]);
    };
    ```
